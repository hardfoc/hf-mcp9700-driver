41template <typename AdcType>
42bool Mcp9700Thermistor<AdcType>::ReadTemperatureCelsius(float* temperature_celsius) noexcept {
43 if (!initialized_ || adc_interface_ == nullptr) {
44 last_error_ = Error::kNotInitialized;
45 return false;
46 }
47 if (temperature_celsius == nullptr) {
48 last_error_ = Error::kInvalidArgument;
49 return false;
50 }
51 float voltage_v = 0.0F;
52 const int err = adc_interface_->ReadChannelV(adc_channel_, &voltage_v);
53 if (err != 0) {
54 last_error_ = Error::kAdcError;
55 return false;
56 }
57 const float celsius = (voltage_v - v_zero_c_) / v_per_c_;
58 // DS20001942 Section 3.2 / Temperature Characteristics: a working sensor's output stays within
59 // the specified temperature range; anything well outside it is a wiring or scaling fault.
60 if (!std::isfinite(celsius) || celsius < min_c_ - margin_c_ || celsius > max_c_ + margin_c_) {
61 last_error_ = Error::kOutOfRange;
62 return false;
63 }
64 *temperature_celsius = celsius;
65 last_error_ = Error::kNone;
66 return true;
67}
